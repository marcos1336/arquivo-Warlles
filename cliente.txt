def mensagem_sucesso(texto):
    print(f"\n✔ {texto}\n")


def formatar_cpf(cpf):
    cpf = cpf.replace(".", "").replace("-", "").strip()

    if len(cpf) == 11 and cpf.isdigit():
        return f"{cpf[:3]}.{cpf[3:6]}.{cpf[6:9]}-{cpf[9:]}"
    
    return cpf


def formatar_telefone(tel):
    tel = tel.replace(" ", "").strip()

    if len(tel) == 11 and tel.isdigit():
        return f"({tel[:2]}) {tel[2:7]}-{tel[7:]}"
    
    return tel


def criar_cliente(id, nome, cpf, telefone, cidade):
    return {
        "id": id,
        "nome": nome,
        "cpf": cpf,
        "telefone": telefone,
        "cidade": cidade
    }


def adicionar_cliente(lista_clientes):
    print("\n=== NOVO CLIENTE ===")

    id = len(lista_clientes) + 1
    nome = input("Nome: ")
    cpf = formatar_cpf(input("CPF: "))
    telefone = formatar_telefone(input("Telefone: "))
    cidade = input("Cidade: ")

    lista_clientes.append(criar_cliente(id, nome, cpf, telefone, cidade))

    mensagem_sucesso(f"Cliente {nome} adicionado com sucesso!")


def excluir_cliente_por_id(lista_clientes):
    print("\n=== EXCLUIR CLIENTE ===")

    try:
        id_busca = int(input("Digite o ID do cliente: "))
    except ValueError:
        print("ID inválido!")
        return

    for i, c in enumerate(lista_clientes):
        if c["id"] == id_busca:
            nome = c["nome"]
            del lista_clientes[i]

            # reorganiza IDs
            for idx, cliente in enumerate(lista_clientes):
                cliente["id"] = idx + 1

            mensagem_sucesso(f"Cliente {nome} removido com sucesso!")
            return

    print("Cliente não encontrado!")

    for c in lista_clientes:

    print("╚════╩════════════════════╩═════════════════╩══════════════╩══════════════════════════╝\n")


def buscar_cliente_por_id(lista_clientes):
    print("\n=== BUSCAR CLIENTE POR ID ===")

    try:
        id_busca = int(input("Digite o ID: "))
    except ValueError:
        print("ID inválido!")
        return

    for c in lista_clientes:
        if c["id"] == id_busca:
            print("\nCliente encontrado:")

            print("║ ID ║ NOME               ║ CPF             ║ TELEFONE     ║ CIDADE                  ║")

            print("╚════╩════════════════════╩═════════════════╩══════════════╩══════════════════════════╝\n")

def salvar_clientes(lista_clientes):

        arquivo.write("║ ID ║ NOME               ║ CPF             ║ TELEFONE     ║ CIDADE                  ║\n")
        arquivo.write("╠════╬════════════════════╬═════════════════╬══════════════╬══════════════════════════╣\n")

        for c in lista_clientes:
            linha = f"║ {c['id']:<2} ║ {c['nome']:<18} ║ {c['cpf']:<15} ║ {c['telefone']:<12} ║ {c['cidade']:<24} ║\n"
            arquivo.write(linha)

        arquivo.write("╚════╩════════════════════╩═════════════════╩══════════════╩══════════════════════════╝\n")

    mensagem_sucesso("Arquivo salvo com sucesso!")
